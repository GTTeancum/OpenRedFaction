cmake_minimum_required(VERSION 3.20)
project(RedFactionReconstruction LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

target_sources(rf_core PRIVATE src/core/model.c)

if(MSVC)
    target_compile_definitions(rf_core PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(rf_core PRIVATE /W4)
else()
    target_compile_options(rf_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
